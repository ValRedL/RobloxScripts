local function GetSafeTime()
    local ok, t = pcall(function() return workspace:GetServerTimeNow() end)
    if ok and typeof(t) == "number" and t > 0 then return t end
    return os.time()
end

local function GetSafeState()
    local ok, s = pcall(function() return StateController:Get() end)
    if ok and typeof(s) == "table" then return s end
    return {}
end

local function SafeRiftStatus(nowTime)
    nowTime = nowTime or GetSafeTime()
    local ok, st = pcall(function() return RiftConfig.Status(nowTime) end)
    if ok and typeof(st) == "table" then return st end
    return { joinable = false, open = false, nextOpenAt = nowTime + 600, joinEndsAt = nowTime + 300 }
end

local function SafeClock(sec)
    if not sec or sec <= 0 then return "0:00" end
    local ok, res = pcall(function() return RiftConfig.Clock(sec) end)
    if ok and typeof(res) == "string" then return res end
    local m = math.floor(sec / 60)
    local s = math.floor(sec % 60)
    return string.format("%d:%02d", m, s)
end

local function SafeStoryCooldown(cooldowns, stageId, nowTime)
    nowTime = nowTime or GetSafeTime()
    local ok, cd = pcall(function()
        return StoryRules.CooldownLeft(cooldowns or {}, stageId, nowTime)
    end)
    if ok and typeof(cd) == "number" then return cd end
    return 0
end

local function LoadConfig()
    if typeof(readfile) == "function" and typeof(isfile) == "function" then
        local ok, exists = pcall(isfile, CONFIG_FILE)
        if ok and exists then
            local readOk, content = pcall(readfile, CONFIG_FILE)
            if readOk and content and #content > 0 then
                local decodeOk, loaded = pcall(function()
                    return HttpService:JSONDecode(content)
                end)
                if decodeOk and typeof(loaded) == "table" then
                    for k, v in pairs(loaded) do
                        if Settings[k] ~= nil then
                            if typeof(v) == "table" and typeof(Settings[k]) == "table" then
                                for subK, subV in pairs(v) do
                                    Settings[k][subK] = subV
                                end
                            else
                                Settings[k] = v
                            end
                        end
                    end
                end
            end
        end
    end
end

local LastTeamSwitch = 0
local function SwitchTeam(targetTeam)
    local state = GetSafeState()
    local current = state and state.ActiveLoadout
    if current ~= targetTeam then
        local now = os.clock()
        if now - LastTeamSwitch > 2.0 then
            LastTeamSwitch = now
            SetStatus(string.format("Switching to Team %d...", targetTeam))
            pcall(function()
                StateController:SetLoadout(targetTeam)
            end)
            task.wait(0.4)
        end
    end
end

local LastAutoBattleCheck = 0
local function EnsureAutoBattle()
    if not Settings.AutoBattle then return end
    local now = os.clock()
    if now - LastAutoBattleCheck < 3.0 then return end
    LastAutoBattleCheck = now

    pcall(function()
        if SettingsController:Get("AutoBattle") ~= true then
            SettingsController:Set("AutoBattle", true)
        end
        if BattleController.SetAuto and not BattleController.AutoOn() then
            BattleController.SetAuto(true)
        end
    end)
end

    end)
end

-- MultiBattle Background Inspection
local function GetBgBattle()
    local attr = LocalPlayer:GetAttribute(MultiBattle.ATTRIBUTE)
    if typeof(attr) == "string" and attr ~= "" then
        local ok, data = pcall(HttpService.JSONDecode, HttpService, attr)
        if ok and typeof(data) == "table" and data.side ~= "over" then
            return data
        end
    end
    return nil
end

local function IsTowerBattleActive()
    if BattleController.IsActive() and BattleController.IsTower() then
        return true
    end
    local bg = GetBgBattle()
    if bg and MultiBattle.IsTower(bg.mode) then
        return true
    end
    return false
end

local function IsRotatorBattleActive()
    if BattleController.IsActive() and not BattleController.IsTower() then
        return true
    end
    local bg = GetBgBattle()
    if bg and not MultiBattle.IsTower(bg.mode) then
        return true
    end
    return false
end

-- 🎯 SMART STORY STAGE SELECTOR (Cycle Mobs + Secret Boss Priority)
-- =====================================================================

local StoryCycleCursor = 1

local function GetNextStoryTarget(targetWorld, targetDiff)
    local worldId = targetWorld or Settings.SelectedWorld
    local diff = targetDiff or Settings.SelectedDifficulty
    local state = GetSafeState()
    local now = GetSafeTime()
    local worldData = WorldMapLookup[worldId] or WorldMapLookup["toy_kingdom"]

    -- 1. HIGHEST PRIORITY: Secret Boss (Ruffy 4 in Toy Kingdom)
    if Settings.StorySecretBoss and worldId == "toy_kingdom" then
        local secId = "toy_kingdom_secret"
        local secCd = SafeStoryCooldown(state.StoryCooldowns, secId, now)
        if secCd <= 0 then
            return secId, "Secret", "Secret Boss: Ruffy 4"
        end
    end

    -- 2. Single Stage Mode
    if Settings.StoryCycleMode == "Single Stage" then
        local stgName = (worldData.StageNames and worldData.StageNames[Settings.SelectedStage]) or Settings.SelectedStage
        return Settings.SelectedStage, diff, stgName
    end

    -- 3. Cycle Mode: Collect all checked stages for this world
    local enabledStages = {}
    for _, sId in ipairs(worldData.Stages) do
        if not sId:match("_secret$") then
            if Settings.StorySelectedStages[sId] == true then
                table.insert(enabledStages, sId)
            end
        end
    end

    if #enabledStages == 0 then
        -- Default to Stage 1 if nothing checked
        local defStage = worldId .. "_1"
        return defStage, diff, (worldData.StageNames and worldData.StageNames[defStage]) or defStage
    end

    -- Cycle to the next ready stage
    for offset = 0, #enabledStages - 1 do
        local checkIdx = ((StoryCycleCursor + offset - 1) % #enabledStages) + 1
        local candidate = enabledStages[checkIdx]
        local cd = SafeStoryCooldown(state.StoryCooldowns, candidate, now)
        if cd <= 0 then
            StoryCycleCursor = (checkIdx % #enabledStages) + 1
            local friendlyName = (worldData.StageNames and worldData.StageNames[candidate]) or candidate
            return candidate, diff, friendlyName
        end
    end

    -- If all enabled stages have cooldown (e.g. only boss checked and on CD)
    local fallback = enabledStages[1]
    local fallbackName = (worldData.StageNames and worldData.StageNames[fallback]) or fallback
    return fallback, diff, fallbackName
end

task.spawn(function()
    while true do
        task.wait(0.8)
        pcall(function()
            if Settings.MultiBattleSystem and Settings.MultiSlot1Enabled then
            if not IsTowerBattleActive() then
                SwitchTeam(Settings.MultiSlot1Team)
                local towerName = Settings.MultiSlot1Tower
                SetStatus(string.format("Multi [Slot 1]: Checking %s Tower...", towerName))

                local stateRes = RemoteTowerAction:InvokeServer("State")
                if stateRes and stateRes.ok and stateRes.state then
                    local towerState = stateRes.state.towers and stateRes.state.towers[towerName] or {}
                    local run = towerState.run
                    
                    if run then
                        SetStatus(string.format("Multi [Slot 1]: Resuming %s Tower...", towerName))
                        RemoteTowerAction:InvokeServer("Resume", towerName)
                        EnsureAutoBattle()
                        task.wait(2)
                    elseif Settings.MultiSlot1Restart then
                        SetStatus(string.format("Multi [Slot 1]: Starting %s Tower...", towerName))
                        local startRes = RemoteTowerAction:InvokeServer("Start", towerName)
                        if startRes and startRes.ok then
                            EnsureAutoBattle()
                            task.wait(2)
                        else
                            task.wait(2.5)
                        end
                    end
                else
                    task.wait(2)
                end
            else
                EnsureAutoBattle()
            end
        end
        end)
    end
end)

                                    end
                                else
                                    SetStatus(string.format("Multi [Slot 2]: %s on CD (%ds)...", curStep.Name, math.ceil(cd)))
                                    if cd > 60 then
                                        CustomPipelineIndex = CustomPipelineIndex + 1
                                    else
                                        task.wait(1.5)
                                    end
                                end
                            else
                                -- Ready to fight Step!
                                SwitchTeam(Settings.StoryTeam)
                                SetStatus(string.format("Multi [Slot 2]: Step %d/5 -> %s [%s]...", CustomPipelineIndex, curStep.Name, curStep.Diff))
                                local sRes = RemoteStoryFight:InvokeServer(curStep.Id, curStep.Diff)
                                if sRes and sRes.ok then
                                    CustomPipelineIndex = CustomPipelineIndex + 1
                                    EnsureAutoBattle()
                                    task.wait(2.5)
                                else
                                    task.wait(2.5)
                                end
                            end
                        end
                    end
                else
                    -- =================================================================
                    -- MODE B: STANDARD FLEXIBLE ROTATOR
                    -- =================================================================
                    local targetStage, targetDiff, friendlyName = GetNextStoryTarget(Settings.SelectedWorld, Settings.SelectedDifficulty)
                    local stageCd = SafeStoryCooldown(state.StoryCooldowns, targetStage, now)

                    local maxQuota = math.huge
                    if Settings.MultiSlot2StoryMode == "Quota 5 Battles" then maxQuota = 5
                    elseif Settings.MultiSlot2StoryMode == "Quota 10 Battles" then maxQuota = 10 end
                    local storyQuotaReached = (MultiSlot2QuotaDone >= maxQuota)

                    local shouldDoRift = false
                    local shouldDoStory = false

                    if Settings.MultiSlot2Strategy == "Rifts (When Open) -> Story" then
                        if riftAvailable then shouldDoRift = true else shouldDoStory = true end
                    else
                        if Settings.MultiSlot2StoryMode == "Cycle Mobs + Secret Boss" then
                            if targetStage:match("_secret$") then shouldDoStory = true
                            elseif stageCd <= 0 then shouldDoStory = true
                            elseif riftAvailable then shouldDoRift = true
                            else shouldDoStory = true end
                        elseif Settings.MultiSlot2StoryMode == "Boss CD Loop" then
                            if stageCd <= 0 then shouldDoStory = true
                            elseif riftAvailable then shouldDoRift = true
                            else shouldDoStory = true end
                        else
                            if not storyQuotaReached then shouldDoStory = true
                            elseif riftAvailable then shouldDoRift = true
                            else MultiSlot2QuotaDone = 0; shouldDoStory = true end
                        end
                    end

                    if shouldDoRift and riftAvailable then
                        if not riftInside then
                            SetStatus("Multi [Slot 2]: Entering Rift Portal...")
                            RemoteRiftAction:InvokeServer("Enter")
                            task.wait(2)
                        else
                            local targetRiftBoss = Settings.MultiSlot2RiftBoss
                            if Settings.MultiSlot2RiftMode == "Auto Progress (1 -> 5)" then
                                local rState = RemoteRiftAction:InvokeServer("State")
                                if rState and rState.ok and typeof(rState.beaten) == "table" then
                                    for bIdx = 1, #RiftBosses do
                                        if not rState.beaten[bIdx] then
                                            targetRiftBoss = bIdx
                                            break
                                        end
                                    end
                                end
                            end

                            SwitchTeam(Settings.MultiSlot2RiftTeam)
                            local bName = RiftBosses[targetRiftBoss] and RiftBosses[targetRiftBoss].Name or tostring(targetRiftBoss)
                            SetStatus(string.format("Multi [Slot 2]: Rifts vs %s...", bName))
                            pcall(function()
                                SettingsController:Set("AutoRift", Settings.MultiSlot2RiftAutoRepeat)
                            end)
                            local fightRes = RemoteRiftAction:InvokeServer("Fight", targetRiftBoss)
                            if fightRes and fightRes.ok then
                                EnsureAutoBattle()
                                task.wait(2.5)
                            else
                                task.wait(3)
                            end
                        end
                    elseif shouldDoStory then
                        if stageCd > 0 and not targetStage:match("_secret$") and not riftAvailable then
                            SetStatus(string.format("Multi [Slot 2]: CD %ds (%s)", math.ceil(stageCd), friendlyName))
                            task.wait(1)
                        else
                            SwitchTeam(Settings.StoryTeam)
                            SetStatus(string.format("Multi [Slot 2]: %s [%s]...", friendlyName, targetDiff))
                            local sRes = RemoteStoryFight:InvokeServer(targetStage, targetDiff)
                            if sRes and sRes.ok then
                                MultiSlot2QuotaDone = MultiSlot2QuotaDone + 1
                                EnsureAutoBattle()
                                task.wait(2.5)
                            else
                                task.wait(2.5)
                            end
                        end
                    end
                end
            else
                EnsureAutoBattle()
            end
        end
        end)
    end
end)

-- Standalone Endgame Loop
task.spawn(function()
    while true do
        task.wait(0.5)
        if not Settings.MultiBattleSystem and Settings.EndgameFarmEnabled then
            local state = GetSafeState()
            local now = GetSafeTime()

            local activeTargets = {}
            for bossId, isEnabled in pairs(Settings.EndgameTargets) do
                if isEnabled then table.insert(activeTargets, bossId) end
            end

            local readyBoss = nil
            local minCooldown = 999999
            for _, bossId in ipairs(activeTargets) do
                local cd = SafeStoryCooldown(state.StoryCooldowns, bossId, now)
                if cd <= 0 then
                    readyBoss = bossId
                    break
                else
                    if cd < minCooldown then minCooldown = cd end
                end
            end

            if readyBoss then
                if BattleController.IsActive() then
                    EnsureAutoBattle()
                    if BattleController.IsTower() then
                        pcall(function() RemoteTowerAction:InvokeServer("Pause", Settings.EndgameTower) end)
                        while BattleController.IsActive() and Settings.EndgameFarmEnabled do
                            task.wait(0.5)
                            EnsureAutoBattle()
                        end
                        task.wait(1.5)
                    else
                        while BattleController.IsActive() and Settings.EndgameFarmEnabled do
                            task.wait(0.5)
                            EnsureAutoBattle()
                        end
                        task.wait(1)
                    end
                else
                    SwitchTeam(Settings.EndgameBossTeam)
                    local diff = Settings.EndgameDifficulty
                    if readyBoss:match("_secret$") then diff = "Secret" end
                    SetStatus("Endgame: Fighting " .. readyBoss .. " [Team " .. Settings.EndgameBossTeam .. "]")
                    local res = RemoteStoryFight:InvokeServer(readyBoss, diff)
                    if res and res.ok then
                        EnsureAutoBattle()
                        while BattleController.IsActive() and Settings.EndgameFarmEnabled do
                            task.wait(0.5)
                            EnsureAutoBattle()
                        end
                        task.wait(1.5)
                    else
                        task.wait(2)
                    end
                end
            else
                local minMin = math.floor(minCooldown / 60)
                local minSec = math.floor(minCooldown % 60)
                if BattleController.IsActive() then
                    EnsureAutoBattle()
                    while BattleController.IsActive() and Settings.EndgameFarmEnabled do
                        task.wait(0.5)
                        EnsureAutoBattle()
                    end
                    task.wait(1.5)
                else
                    SwitchTeam(Settings.EndgameTowerTeam)
                    SetStatus(string.format("Endgame: %s (Boss in %dm %02ds)", Settings.EndgameTower, minMin, minSec))
                    local stateRes = RemoteTowerAction:InvokeServer("State")
                    if stateRes and stateRes.ok and stateRes.state then
                        local towerState = stateRes.state.towers and stateRes.state.towers[Settings.EndgameTower] or {}
                        if towerState.run then
                            RemoteTowerAction:InvokeServer("Resume", Settings.EndgameTower)
                        else
                            RemoteTowerAction:InvokeServer("Start", Settings.EndgameTower)
                        end
                        task.wait(2)
                    else
                        task.wait(2)
                    end
                end
            end
        end
    end
end)

local function UpdateDualMonitorUI()
    local ok, err = pcall(function()
        local towerAct = IsTowerBattleActive()
        local rotAct = IsRotatorBattleActive()
        local now = GetSafeTime()
        local s = GetSafeState()
        local status = SafeRiftStatus(now)

        DmSlot1.Text = string.format("Slot 1 [Tower]: %s (%s) [Team %d]", 
            Settings.MultiSlot1Tower, 
            towerAct and "IN BATTLE 🟢" or "Searching/Ready ⏳", 
            Settings.MultiSlot1Team)

        if Settings.MultiSlot2StoryMode == "Custom Pipeline" then
            local curStageName = (CustomPipelineStages[CustomPipelineIndex] and CustomPipelineStages[CustomPipelineIndex].Name) or "Rifts Mode"
            local stepTxt = CustomPipelineIndex <= #CustomPipelineStages and string.format("Step %d/5 (%s)", CustomPipelineIndex, curStageName) or "Rifts Mode"
            DmSlot2.Text = string.format("Slot 2 [Pipeline]: %s (%s)", stepTxt, rotAct and "IN BATTLE 🟢" or "Ready ⏳")
        else
            DmSlot2.Text = string.format("Slot 2 [Rotator]: %s (%s) [Done: %d]", 
                Settings.MultiSlot2StoryMode, 
                rotAct and "IN BATTLE 🟢" or "Ready ⏳", 
                MultiSlot2QuotaDone)
        end

        if status.joinable then
            local left = math.max(0, (status.joinEndsAt or now) - now)
            DmRift.Text = string.format("Rift Portal: OPEN & JOINABLE (%s left) ⚡", SafeClock(left))
        elseif status.open then
            DmRift.Text = string.format("Rift Portal: OPEN (In Progress) 🟢")
        else
            local nextIn = math.max(0, (status.nextOpenAt or (now + 600)) - now)
            DmRift.Text = string.format("Rift Portal: Closed (Next open in %s) ⏳", SafeClock(nextIn))
        end

        local cdSec = SafeStoryCooldown(s.StoryCooldowns, "toy_kingdom_secret", now)
        if cdSec <= 0 then
            DmSecret.Text = "Secret Boss (Ruffy 4): READY TO FIGHT! 🟢"
            DmSecret.TextColor3 = Color3.fromRGB(46, 204, 113)
        else
            DmSecret.Text = string.format("Secret Boss (Ruffy 4): CD %s ⏳", SafeClock(cdSec))
            DmSecret.TextColor3 = Color3.fromRGB(255, 150, 180)
        end
    end)
    if not ok then
        DmSlot1.Text = 'Monitor error: ' .. tostring(err)
        DmSlot1.TextWrapped = true
        DmSlot2.Text = 'Status update failed; see error above'
        DmRift.Text = 'Rift Portal: status unavailable'
        DmSecret.Text = 'Secret Boss: status unavailable'
    end
end

-- Run immediately upon UI creation so labels NEVER stay on 'Checking...'
task.spawn(UpdateDualMonitorUI)

task.spawn(function()
    while true do
        task.wait(1)
        UpdateDualMonitorUI()
    end
end)
