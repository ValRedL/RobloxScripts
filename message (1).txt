local RiftSnapshot = nil
local RiftSnapshotAt = -math.huge
local RiftCheckPending = false
local RiftCheckStarted = 0
local RiftNextCheck = 0
local RiftCheckError = nil

local function SafeRiftStatus(nowTime)
    nowTime = nowTime or GetSafeTime()
    local tickNow = os.clock()
    if not RiftCheckPending and tickNow >= RiftNextCheck then
        RiftCheckPending = true
        RiftCheckStarted = tickNow
        task.defer(function()
            local ok, result = pcall(function() return RiftConfig.Status(nowTime) end)
            local finishedAt = os.clock()
            RiftCheckPending = false
            RiftNextCheck = finishedAt + (ok and 2 or 10)
            if ok and typeof(result) == "table"
                and typeof(result.joinable) == "boolean"
                and typeof(result.open) == "boolean"
                and typeof(result.nextOpenAt) == "number" then
                RiftSnapshot = result
                -- A delayed result describes the request time, not completion time.
                RiftSnapshotAt = tickNow
                RiftCheckError = nil
            else
                RiftSnapshot = nil
                RiftCheckError = ok and "Invalid Rift status" or tostring(result)
                RiftNextCheck = finishedAt + 10
            end
        end)
    end
    if RiftSnapshot and tickNow - RiftSnapshotAt <= 5 then
        local result = table.clone(RiftSnapshot)
        if result.joinable and typeof(result.joinEndsAt) == "number" and nowTime >= result.joinEndsAt then
            result.joinable = false
        end
        if result.open and typeof(result.closesAt) == "number" and nowTime >= result.closesAt then
            result.open = false
            result.joinable = false
        end
        return result
    end
    local reason = RiftCheckError or "Reading Rift status"
    if RiftCheckPending and tickNow - RiftCheckStarted >= 5 then
        reason = "RiftConfig.Status not responding; Story/Tower can continue"
    end
    return {available = false, reason = reason, joinable = false, open = false,
        nextOpenAt = nowTime + 5, joinEndsAt = nowTime}
end

local function SafeClock(sec)
    local n = tonumber(sec) or 0
    if n ~= n or n == math.huge or n <= 0 then return "0:00" end
    local total = math.ceil(n)
    return string.format("%d:%02d", math.floor(total / 60), total % 60)
end

local function SafeStoryCooldown(cooldowns, stageId, nowTime)
    nowTime = nowTime or GetSafeTime()
    local ok, cd = pcall(function()
        return StoryRules.CooldownLeft(cooldowns or {}, stageId, nowTime)
    end)
    if ok and typeof(cd) == "number" then return cd end
    return 0
end

local function ResolveSlot1Tower(selected, serverTowers)

    for id in pairs(serverTowers) do
        if typeof(id) == "string" then
            local lower = string.lower(id)
            if string.find(lower, "halloween", 1, true) or string.find(lower, "haloween", 1, true) then
                table.insert(matches, id)
            end
        end
    end
    if #matches == 1 then return matches[1] end
    if #matches > 1 then return nil, "Multiple Halloween towers found; exact ID required" end
    return nil, "Halloween Tower not listed by server; try entering it manually once"
end

local StoryRetryAt = {}
local LastStoryFailure = ""
local function StoryTargetReady(stageId)
    return os.clock() >= (StoryRetryAt[stageId] or 0)
end
local function TryStoryFight(stageId, difficulty)
    if not StoryTargetReady(stageId) then
        return {ok = false, reason = LastStoryFailure, retryPending = true}
    end
    local ok, result = pcall(function()
        return RemoteStoryFight:InvokeServer(stageId, difficulty)
    end)
    if ok and typeof(result) == "table" and result.ok then
        StoryRetryAt[stageId] = nil
        LastStoryFailure = ""
        return result
    end
    local reason = ok and (typeof(result) == "table" and result.reason or "No valid server response") or result
    LastStoryFailure = tostring(stageId) .. ": " .. tostring(reason)
    StoryRetryAt[stageId] = os.clock() + (ok and 60 or 10)
    SetStatus("Story rejected: " .. LastStoryFailure)
    warn("[ACB Story] " .. LastStoryFailure)
    return {ok = false, reason = tostring(reason)}
end

local function NormalizeSlot2Settings()
    local function boss(value, fallback)
        local number = tonumber(value)
        if not number or number ~= number then return fallback end
        return math.clamp(math.floor(number), 1, 5)
    end
    local function teamNum(value, fallback)
        local number = tonumber(value)
        if not number or number ~= number then return fallback end
        return math.clamp(math.floor(number), 1, 3)
    end
    Settings.MultiSlot2RiftRepeatBoss = boss(Settings.MultiSlot2RiftRepeatBoss, 4)
    Settings.MultiSlot2RiftBoss = boss(Settings.MultiSlot2RiftBoss, 1)
    if typeof(Settings.MultiSlot2SmartCycle) ~= "boolean" then Settings.MultiSlot2SmartCycle = true end
    if typeof(Settings.MultiSlot2ReturnStory) ~= "boolean" then Settings.MultiSlot2ReturnStory = true end
    if typeof(Settings.StoryCloverSecretBoss) ~= "boolean" then Settings.StoryCloverSecretBoss = true end
    if typeof(Settings.MultiSlot3Enabled) ~= "boolean" then Settings.MultiSlot3Enabled = false end
    if not table.find(Slot3Modes, Settings.MultiSlot3Mode) then Settings.MultiSlot3Mode = "2nd Tower" end
    if not table.find(Slot1Towers, Settings.MultiSlot3Tower) then Settings.MultiSlot3Tower = "Infinite" end
    Settings.MultiSlot3Team = teamNum(Settings.MultiSlot3Team, 3)
    if typeof(Settings.MultiSlot3Restart) ~= "boolean" then Settings.MultiSlot3Restart = true end
    if typeof(Settings.StorySelectedStages) ~= "table" then Settings.StorySelectedStages = {} end
    for _, cloverStageId in ipairs({"clover_capital_1", "clover_capital_2", "clover_capital_3", "clover_capital_4", "clover_capital_boss"}) do
        if typeof(Settings.StorySelectedStages[cloverStageId]) ~= "boolean" then
            Settings.StorySelectedStages[cloverStageId] = true
        end
    end
end

end

local function LoadConfig()
    if typeof(readfile) == "function" and typeof(isfile) == "function" then
        local ok, exists = pcall(isfile, CONFIG_FILE)
        if ok and exists then
            local readOk, content = pcall(readfile, CONFIG_FILE)
            if readOk and content and #content > 0 then
                local decodeOk, loaded = pcall(function()
                    return HttpService:JSONDecode(content)
                end)
                if decodeOk and typeof(loaded) == "table" then
                    for k, v in pairs(loaded) do
                        if Settings[k] ~= nil then
                            if typeof(v) == "table" and typeof(Settings[k]) == "table" then
                                for subK, subV in pairs(v) do
                                    Settings[k][subK] = subV
                                end
                            else
                                Settings[k] = v
                            end
                        end
                    end
                end
            end
        end
    end
end

local LastTeamSwitch = 0
local LastTeamSwitchTarget = nil
local function SwitchTeam(targetTeam)
    local state = GetSafeState()
    local current = state and state.ActiveLoadout
    if current ~= targetTeam then
        local now = os.clock()
        if targetTeam ~= LastTeamSwitchTarget or (now - LastTeamSwitch > 2.0) then
            LastTeamSwitch = now
            LastTeamSwitchTarget = targetTeam
            SetStatus(string.format("Switching to Team %d...", targetTeam))
            pcall(function()
                StateController:SetLoadout(targetTeam)
            end)
            task.wait(0.4)
        end
    end
end

local ActionLockBusy = false
local function WithActionLock(fn)
    local deadline = os.clock() + 8
    while ActionLockBusy and os.clock() < deadline do
        task.wait(0.1)
    end
    if ActionLockBusy then
        return false, "Action lock busy"
    end
    ActionLockBusy = true
    local ok, r1, r2, r3 = pcall(fn)
    ActionLockBusy = false
    if not ok then
        return false, r1
    end
    return true, r1, r2, r3
end

local LastAutoBattleCheck = 0
local function EnsureAutoBattle()
    if not Settings.AutoBattle then return end
    local now = os.clock()
    if now - LastAutoBattleCheck < 3.0 then return end
    LastAutoBattleCheck = now

    pcall(function()
        if SettingsController:Get("AutoBattle") ~= true then
            SettingsController:Set("AutoBattle", true)
        end
        if BattleController.SetAuto and not BattleController.AutoOn() then
            BattleController.SetAuto(true)
        end
    end)
end

local function DecodeBgAttr(attrName)
    if typeof(attrName) ~= "string" or attrName == "" then return nil end
    local okAttr, attr = pcall(function() return LocalPlayer:GetAttribute(attrName) end)
    if okAttr and typeof(attr) == "string" and attr ~= "" then
        local ok, data = pcall(HttpService.JSONDecode, HttpService, attr)
        if ok and typeof(data) == "table" and data.side ~= "over" then
            return data
        end
    end
    return nil
end

local function GetBgMode(bg)
    if typeof(bg) ~= "table" then return nil end
    if typeof(bg.mode) == "string" and bg.mode ~= "" then return bg.mode end
    if typeof(bg.tower) == "string" and bg.tower ~= "" then
        if MultiBattle.TowerMode then return MultiBattle.TowerMode(bg.tower) end
        return "tower:" .. bg.tower
    end
    return nil
end

local function GetForegroundMode()
    local okActive, active = pcall(function() return BattleController.IsActive() end)
    if not okActive or not active then return nil end
    local okShown, shown = pcall(function() return BattleController.ShownMode and BattleController.ShownMode() end)
    if okShown and typeof(shown) == "string" and shown ~= "" then return shown end
    local okHidden, hidden = pcall(function() return BattleController.HiddenMode and BattleController.HiddenMode() end)
    if okHidden and typeof(hidden) == "string" and hidden ~= "" then return hidden end
    local okTower, isTower = pcall(function() return BattleController.IsTower and BattleController.IsTower() end)
    if okTower and isTower then return "tower" end
    return "other"
end

local function IsModeBattleActive(targetMode)
    if typeof(targetMode) ~= "string" or targetMode == "" then return false end
    if GetForegroundMode() == targetMode then
        return true
    end
    for _, bg in ipairs(GetAllBgBattles()) do
        if GetBgMode(bg) == targetMode then
            return true
        end
    end
    return false
end

local function GetTowerBackgroundWaitId()
    local okMod, towerCtrl = pcall(function() return require(Client.Controllers.TowerController) end)
    if not okMod or typeof(towerCtrl) ~= "table" or typeof(towerCtrl.BackgroundWait) ~= "function" then
        return nil
    end
    local okWait, waitInfo = pcall(towerCtrl.BackgroundWait)
    if okWait and typeof(waitInfo) == "table" and typeof(waitInfo.id) == "string" then
        return waitInfo.id
    end
    return nil
end

local function IsTowerBattleActive(specificTowerId)
    local fgMode = GetForegroundMode()
    local expectedMode = specificTowerId and (MultiBattle.TowerMode and MultiBattle.TowerMode(specificTowerId) or ("tower:" .. specificTowerId)) or nil

    if fgMode then
        local okIsTower, fgIsTower = pcall(function() return BattleController.IsTower() end)
        if not specificTowerId then
            if (okIsTower and fgIsTower) or MultiBattle.IsTower(fgMode) then
                return true
            end
        else
            if fgMode == expectedMode or (fgMode == "tower" and okIsTower and fgIsTower) then
                return true
            end
        end
    end

    for _, bg in ipairs(GetAllBgBattles()) do
        local bgMode = GetBgMode(bg)
        if not specificTowerId then
            if MultiBattle.IsTower(bgMode) then
                return true
            end
        elseif bgMode == expectedMode or bg.tower == specificTowerId then
            return true
        end
    end

    local bwId = GetTowerBackgroundWaitId()
    if bwId and (not specificTowerId or bwId == specificTowerId) then
        return true
    end

    return false
end

local function IsStoryBattleActive()
    if IsModeBattleActive("story") then return true end
    local fgMode = GetForegroundMode()
    if fgMode == "other" and not BattleController.ShownMode then
        return true
    end
    return false
end

end

    return IsStoryBattleActive() or IsRiftBattleActive()
end

local StoryCycleCursor = 1

local function GetReadySecretBoss(state, now)
    if Settings.StoryCloverSecretBoss and StoryTargetReady("clover_capital_secret") then
        local cd = SafeStoryCooldown(state.StoryCooldowns, "clover_capital_secret", now)
        if cd <= 0 then
            return "clover_capital_secret", "Secret", "Clover Throne Room: Time Sage King"
        end
    end
    if Settings.StorySecretBoss and StoryTargetReady("toy_kingdom_secret") then
        local cd = SafeStoryCooldown(state.StoryCooldowns, "toy_kingdom_secret", now)
        if cd <= 0 then
            return "toy_kingdom_secret", "Secret", "Secret Boss: Ruffy 4"
        end
    end
    return nil
end

local function GetNextStoryTarget(targetWorld, targetDiff, peek)
    local worldId = targetWorld or Settings.SelectedWorld
    local diff = targetDiff or Settings.SelectedDifficulty
    local state = GetSafeState()
    local now = GetSafeTime()
    local worldData = WorldMapLookup[worldId] or WorldMapLookup["toy_kingdom"]

    -- 1. HIGHEST PRIORITY: Secret Bosses (Clover Throne Room + Toy Kingdom)
    local secId, secDiff, secName = GetReadySecretBoss(state, now)
    if secId then
        return secId, secDiff, secName
    end

    -- 2. Single Stage Mode
    if Settings.StoryCycleMode == "Single Stage" then
        local stgName = (worldData.StageNames and worldData.StageNames[Settings.SelectedStage]) or Settings.SelectedStage
        return Settings.SelectedStage, diff, stgName
    end

    -- 3. Cycle Mode: Collect all checked stages for this world
    local enabledStages = {}
    for _, sId in ipairs(worldData.Stages) do
        if not sId:match("_secret$") then
            if Settings.StorySelectedStages[sId] == true then
                table.insert(enabledStages, sId)
            end
        end
    end

    if #enabledStages == 0 then
        -- Default to Stage 1 if nothing checked
        local defStage = worldId .. "_1"
        return defStage, diff, (worldData.StageNames and worldData.StageNames[defStage]) or defStage
    end

    -- Cycle to the next ready stage
    for offset = 0, #enabledStages - 1 do
        local checkIdx = ((StoryCycleCursor + offset - 1) % #enabledStages) + 1
        local candidate = enabledStages[checkIdx]
        local cd = SafeStoryCooldown(state.StoryCooldowns, candidate, now)
        if cd <= 0 and StoryTargetReady(candidate) then
            if not peek then StoryCycleCursor = (checkIdx % #enabledStages) + 1 end
            local friendlyName = (worldData.StageNames and worldData.StageNames[candidate]) or candidate
            return candidate, diff, friendlyName
        end
    end

    -- If all enabled stages have cooldown (e.g. only boss checked and on CD)
    local fallback = enabledStages[1]
    local fallbackName = (worldData.StageNames and worldData.StageNames[fallback]) or fallback
    return fallback, diff, fallbackName
end

local function PickRiftBoss(beaten, progress, specific, repeatBoss)
    if not progress then return specific end
    for i = 1, 5 do
        if not beaten[i] and not beaten[tostring(i)] then return i end
    end
    return repeatBoss
end

local function RunRiftSlotStep(opts)
    if not opts.riftInside then
        local res = RemoteRiftAction:InvokeServer("Enter")
        local ok = res and res.ok
        opts.setStatus(ok and "Entering Rift" or ("Rift entry failed: " .. tostring(res and res.reason)))
        if not ok and opts.onEnterFailed then
            opts.onEnterFailed()
        end
        return false, ok and 2.5 or 5
    end

    local progress = opts.useProgress
    local beaten = {}
    if progress then
        local rs = RemoteRiftAction:InvokeServer("State")
        if not rs or not rs.ok or typeof(rs.beaten) ~= "table" then
            opts.setStatus("Rift status unavailable; retrying")
            if opts.onStateFailed then
                opts.onStateFailed()
            end
            return false, 5
        end
        beaten = rs.beaten
    end

    local boss = PickRiftBoss(beaten, progress, opts.specificBoss, opts.repeatBoss)
    local fightRes = nil
    local teamMissing = false
    local lockOk, err = WithActionLock(function()
        SwitchTeam(opts.team)
        if GetSafeState().ActiveLoadout ~= opts.team then
            teamMissing = true
            opts.setStatus("Waiting for Team " .. tostring(opts.team))
            return
        end
        if opts.autoRepeat ~= nil then
            pcall(function()
                SettingsController:Set("AutoRift", opts.autoRepeat)
            end)
        end
        local bName = (RiftBosses[boss] and RiftBosses[boss].Name) or tostring(boss)
        opts.setStatus(string.format("Rifts vs %s...", bName))
        fightRes = RemoteRiftAction:InvokeServer("Fight", boss)
    end)

    if not lockOk then
        opts.setStatus("Rift lock error: " .. tostring(err))
        return false, 2
    end
    if teamMissing then
        return false, 1
    end
    if fightRes and fightRes.ok then
        EnsureAutoBattle()
        return true, 2.5, boss
    end
    opts.setStatus("Rift fight failed: " .. tostring(fightRes and fightRes.reason))
    return false, 3, boss
end

local ResolvedTowerCache = {}
local function GetCachedTowerId(selected)
    if selected ~= "Halloween Tower" then return selected end
    return ResolvedTowerCache[selected]
end

local function RunTowerSlotStep(slotLabel, selectedTower, teamNumber, autoRestart, conflictTower, setSlotStatus)
    local cached = GetCachedTowerId(selectedTower)
    if cached and IsTowerBattleActive(cached) then
        setSlotStatus(cached .. " active")
        EnsureAutoBattle()
        return 0.8
    end

    setSlotStatus(string.format("Checking %s Tower...", selectedTower))
    local stateRes = RemoteTowerAction:InvokeServer("State")
    if not stateRes or not stateRes.ok or not stateRes.state then
        setSlotStatus("Tower state unavailable")
        return 2
    end

    local resolved, resolveError = ResolveSlot1Tower(selectedTower, stateRes.state.towers)
    if not resolved then
        setSlotStatus(tostring(resolveError))
        return 5
    end
    if selectedTower == "Halloween Tower" then
        ResolvedTowerCache[selectedTower] = resolved
    end

    if conflictTower and resolved == conflictTower then
        setSlotStatus("Conflict: same tower as Slot 1 (" .. tostring(resolved) .. ")")
        return 5
    end

    if IsTowerBattleActive(resolved) then
        setSlotStatus(resolved .. " active")
        EnsureAutoBattle()
        return 0.8
    end

    local towerState = stateRes.state.towers and stateRes.state.towers[resolved] or {}
    local run = towerState.run
    if not run and not autoRestart then
        setSlotStatus(resolved .. " idle (Restart OFF)")
        return 2
    end

    local actionName = run and "Resume" or "Start"
    local actionVerb = run and "Resuming" or "Starting"
    local actionRes = nil
    local lockOk, err = WithActionLock(function()
        pcall(function()
            RemoteTowerAction:InvokeServer("SetDeck", resolved, teamNumber)
        end)
        SwitchTeam(teamNumber)
        setSlotStatus(string.format("%s %s...", actionVerb, resolved))
        actionRes = RemoteTowerAction:InvokeServer(actionName, resolved)
    end)

    if not lockOk then
        setSlotStatus("Lock error: " .. tostring(err))
        return 2
    end
    if actionRes and actionRes.ok then
        EnsureAutoBattle()
        return 2
    end
    setSlotStatus(tostring(actionRes and actionRes.reason or (actionName .. " failed")))
    return 2.5
end

local function ChooseSmartAction(cycle, ready, secretReady, riftAvailable, returnStory)
    if secretReady then return "secret" end
    if cycle.storyPhase then
        for i = cycle.index, #CustomPipelineStages do
            if ready[i] then return "story", i end
        end
        cycle.storyPhase = false
        cycle.index = #CustomPipelineStages + 1
    end
    -- Give Rift one completed fight between Story rounds, even if mobs have no cooldown.
    if not riftAvailable or (returnStory and cycle.riftFought) then
        for i = 1, #CustomPipelineStages do
            if ready[i] then return "story", i end
        end
    end
    if riftAvailable then return "rift" end
    return "wait"
end

local function SyncSmartAutoRepeat()
    if Settings.MultiSlot3Enabled and Settings.MultiSlot3Mode == "Rifts (Dedicated)" then
        return true

    if Settings.MultiSlot2SmartCycle then desired = false end
    if LastRiftAutoSetting ~= desired then
        local ok = pcall(function() SettingsController:Set("AutoRift", desired) end)
        if ok then LastRiftAutoSetting = desired end
        return ok
    end
    return true
end

local function RunSmartSlot2(state, now, riftInside, riftAvailable)
    if os.clock() < SmartNextAttempt then return end
    local ready = {}
    for i, stage in ipairs(CustomPipelineStages) do
        ready[i] = (Settings.StorySelectedStages[stage.Id] ~= false)
            and StoryTargetReady(stage.Id)
            and SafeStoryCooldown(state.StoryCooldowns, stage.Id, now) <= 0
    end
    local secId, secDiff, secName = GetReadySecretBoss(state, now)
    local secretReady = secId ~= nil
    local action, index = ChooseSmartAction(SmartCycle, ready, secretReady,
        riftAvailable, Settings.MultiSlot2ReturnStory)
    if action == "secret" or action == "story" then
        local stage = action == "secret"
            and {Id = secId, Diff = secDiff, Name = secName}
            or CustomPipelineStages[index]
        local res = nil
        local teamMissing = false
        local lockOk, err = WithActionLock(function()
            if not SmartTeamReady(Settings.StoryTeam) then
                teamMissing = true
                return
            end
            SmartSetStatus("Story: " .. stage.Name)
            res = TryStoryFight(stage.Id, stage.Diff)
        end)
        if not lockOk then
            SmartSetStatus("Story lock error: " .. tostring(err))
            SmartNextAttempt = os.clock() + 2
            return
        end
        if teamMissing then return end
        if action == "story" then
            SmartCycle.storyPhase = true
            SmartCycle.index = index + 1
            SmartCycle.riftFought = false
            CustomPipelineIndex = index + 1
        end
        if res and res.ok then
            EnsureAutoBattle()
        else
            SmartSetStatus("Story unavailable: " .. tostring(res and res.reason))
        end
        SmartNextAttempt = os.clock() + 2.5
    elseif action == "rift" then
        local ok, delaySec, boss = RunRiftSlotStep({
            riftInside = riftInside,
            team = Settings.MultiSlot2RiftTeam,
            useProgress = Settings.MultiSlot2RiftMode == "Auto Progress (1 -> 5)",
            specificBoss = Settings.MultiSlot2RiftBoss,
            repeatBoss = Settings.MultiSlot2RiftRepeatBoss,
            autoRepeat = nil,
            setStatus = SmartSetStatus,
            onEnterFailed = function() SmartCycle.riftFought = true end,
            onStateFailed = function() SmartCycle.riftFought = true end,
        })
        if boss then
            SmartCycle.riftFought = true
            if ok then
                SmartSetStatus("Rift Boss " .. tostring(boss) .. "; checking Story after fight")
            end
        end
        SmartNextAttempt = os.clock() + delaySec
    else
        SmartSetStatus("Waiting for Story cooldown / Rift entry")
        SmartNextAttempt = os.clock() + 2
    end
end

                    local waitSec = math.max(0, (riftStatus.nextOpenAt or (now + 5)) - now)
                    Slot3Status = "Rift closed (" .. SafeClock(waitSec) .. ")"
                end
            elseif mode == "Auto Dungeon" then
                pcall(function()
                    if SettingsController:Get("AutoDungeon") ~= true then
                        SettingsController:Set("AutoDungeon", true)
                    end
                end)
                if IsModeBattleActive("dungeon") then
                    Slot3Status = "In Dungeon battle"
                    EnsureAutoBattle()
                    return
                end
                if not RemoteDungeonAction then
                    Slot3SetStatus("DungeonAction remote unavailable")
                    task.wait(5)
                    return
                end
                if os.clock() >= Slot3NextDungeonAttempt then
                    local autoRes = nil
                    WithActionLock(function()
                        autoRes = RemoteDungeonAction:InvokeServer("Auto", true)
                    end)
                    if autoRes and autoRes.ok then
                        Slot3SetStatus("Auto Dungeon active")
                        Slot3NextDungeonAttempt = os.clock() + 8
                    else
                        Slot3SetStatus("Dungeon: " .. tostring(autoRes and autoRes.reason or "Waiting for Auto Dungeon"))
                        Slot3NextDungeonAttempt = os.clock() + 10
                    end
                end
            else
                local raidInfo = Slot3RaidMap[mode]
                if not raidInfo then
                    Slot3Status = "Unknown mode"
                    return
                end
                if IsModeBattleActive(raidInfo.mode) then
                    Slot3Status = "In " .. mode
                    EnsureAutoBattle()
                    return
                end
                if not RemoteRaidFight then
                    Slot3SetStatus("RaidFight remote unavailable")
                    task.wait(5)
                    return
                end
                if os.clock() >= Slot3NextRaidAttempt then
                    local fightRes = nil
                    local lockOk, lockErr = WithActionLock(function()
                        pcall(function()
                            RemoteRaidFight:InvokeServer("SetDeck", Settings.MultiSlot3Team, raidInfo.id)
                        end)
                        SwitchTeam(Settings.MultiSlot3Team)
                        Slot3SetStatus("Starting " .. mode .. "...")
                        fightRes = RemoteRaidFight:InvokeServer("Fight", raidInfo.id)
                    end)
                    if not lockOk then
                        Slot3SetStatus("Raid lock error: " .. tostring(lockErr))
                        Slot3NextRaidAttempt = os.clock() + 3
                        return
                    end
                    if fightRes and fightRes.ok then
                        Slot3SetStatus("Fighting " .. mode)
                        EnsureAutoBattle()
                        Slot3NextRaidAttempt = os.clock() + 3
                    else
                        local reason = tostring(fightRes and fightRes.reason or "Not ready")
                        Slot3SetStatus(mode .. ": " .. reason)
                        Slot3NextRaidAttempt = os.clock() + 8
                    end
                end
            end
        end)
        if not ok then
            Slot3Status = "Error: " .. tostring(err)
        end
    end
end)

-- Standalone Endgame Loop
task.spawn(function()
    while true do
        task.wait(0.5)
        if not Settings.MultiBattleSystem and Settings.EndgameFarmEnabled then
            local state = GetSafeState()
            local now = GetSafeTime()

            local activeTargets = {}
            for bossId, isEnabled in pairs(Settings.EndgameTargets) do
                if isEnabled then table.insert(activeTargets, bossId) end
            end

            local readyBoss = nil
            local minCooldown = 999999
            for _, bossId in ipairs(activeTargets) do
                local cd = SafeStoryCooldown(state.StoryCooldowns, bossId, now)
                if cd <= 0 then
                    readyBoss = bossId
                    break
                else
                    if cd < minCooldown then minCooldown = cd end
                end
            end

            if readyBoss then
                if BattleController.IsActive() then
                    EnsureAutoBattle()
                    if BattleController.IsTower() then
                        pcall(function() RemoteTowerAction:InvokeServer("Pause", Settings.EndgameTower) end)
                        while BattleController.IsActive() and Settings.EndgameFarmEnabled do
                            task.wait(0.5)
                            EnsureAutoBattle()
                        end
                        task.wait(1.5)
                    else
                        while BattleController.IsActive() and Settings.EndgameFarmEnabled do
                            task.wait(0.5)
                            EnsureAutoBattle()
                        end
                        task.wait(1)
                    end
                else
                    SwitchTeam(Settings.EndgameBossTeam)
                    local diff = Settings.EndgameDifficulty
                    if readyBoss:match("_secret$") then diff = "Secret" end
                    SetStatus("Endgame: Fighting " .. readyBoss .. " [Team " .. Settings.EndgameBossTeam .. "]")
                    local res = TryStoryFight(readyBoss, diff)
                    if res and res.ok then
                        EnsureAutoBattle()
                        while BattleController.IsActive() and Settings.EndgameFarmEnabled do
                            task.wait(0.5)
                            EnsureAutoBattle()
                        end
                        task.wait(1.5)
                    else
                        task.wait(2)
                    end
                end
            else
                local minMin = math.floor(minCooldown / 60)
                local minSec = math.floor(minCooldown % 60)
                if BattleController.IsActive() then
                    EnsureAutoBattle()
                    while BattleController.IsActive() and Settings.EndgameFarmEnabled do
                        task.wait(0.5)
                        EnsureAutoBattle()
                    end
                    task.wait(1.5)
                else
                    SwitchTeam(Settings.EndgameTowerTeam)
                    SetStatus(string.format("Endgame: %s (Boss in %dm %02ds)", Settings.EndgameTower, minMin, minSec))
                    local stateRes = RemoteTowerAction:InvokeServer("State")
                    if stateRes and stateRes.ok and stateRes.state then
                        local towerState = stateRes.state.towers and stateRes.state.towers[Settings.EndgameTower] or {}
                        if towerState.run then
                            RemoteTowerAction:InvokeServer("Resume", Settings.EndgameTower)
                        else
                            RemoteTowerAction:InvokeServer("Start", Settings.EndgameTower)
                        end
                        task.wait(2)
                    else
                        task.wait(2)
                    end
                end
            end
        end
    end
end)

local function UpdateDualMonitorUI()
    local ok, err = pcall(function()
        local slot1TowerId = GetCachedTowerId(Settings.MultiSlot1Tower) or Settings.MultiSlot1Tower
        local towerAct = IsTowerBattleActive(slot1TowerId)
        local rotAct = IsRotatorBattleActive()
        local now = GetSafeTime()
        local s = GetSafeState()
        local status = SafeRiftStatus(now)

        if not Settings.MultiSlot1Enabled then
            DmSlot1.Text = "Slot 1 [Tower]: OFF"
        else
            DmSlot1.Text = string.format("Slot 1 [Tower]: %s (%s) [Team %d]",
                Settings.MultiSlot1Tower,
                towerAct and "IN BATTLE 🟢" or "Searching/Ready ⏳",
                Settings.MultiSlot1Team)
        end

        if not Settings.MultiSlot2Enabled then
            DmSlot2.Text = "Slot 2 [Rotator]: OFF"
        elseif Settings.MultiSlot2SmartCycle then
            DmSlot2.Text = "Slot 2 [Auto]: " .. SmartStatus
        elseif Settings.MultiSlot2StoryMode == "Custom Pipeline" then
            local curStageName = (CustomPipelineStages[CustomPipelineIndex] and CustomPipelineStages[CustomPipelineIndex].Name) or "Rifts Mode"
            local stepTxt = CustomPipelineIndex <= #CustomPipelineStages and string.format("Step %d/%d (%s)", CustomPipelineIndex, #CustomPipelineStages, curStageName) or "Rifts Mode"
            DmSlot2.Text = string.format("Slot 2 [Pipeline]: %s (%s)", stepTxt, rotAct and "IN BATTLE 🟢" or "Ready ⏳")
        else
            DmSlot2.Text = string.format("Slot 2 [Rotator]: %s (%s) [Done: %d]",
                Settings.MultiSlot2StoryMode,
                rotAct and "IN BATTLE 🟢" or "Ready ⏳",
                MultiSlot2QuotaDone)
        end

        if LastStoryFailure ~= "" then
            DmSlot2.Text = "Story rejected (retry in up to 60s): " .. LastStoryFailure
        end

        if not Settings.MultiSlot3Enabled then
            DmSlot3.Text = "Slot 3 [Seal]: OFF"
        else
            local slot3Active = false
            if Settings.MultiSlot3Mode == "2nd Tower" then
                local slot3TowerId = GetCachedTowerId(Settings.MultiSlot3Tower) or Settings.MultiSlot3Tower
                slot3Active = IsTowerBattleActive(slot3TowerId)
            elseif Settings.MultiSlot3Mode == "Rifts (Dedicated)" then
                slot3Active = IsRiftBattleActive()
            elseif Settings.MultiSlot3Mode == "Auto Dungeon" then
                slot3Active = IsModeBattleActive("dungeon")
            elseif Slot3RaidMap[Settings.MultiSlot3Mode] then
                slot3Active = IsModeBattleActive(Slot3RaidMap[Settings.MultiSlot3Mode].mode)
            end
            DmSlot3.Text = string.format("Slot 3 [%s]: %s (%s)",
                Settings.MultiSlot3Mode,
                Slot3Status,
                slot3Active and "IN BATTLE 🟢" or "Ready ⏳")
        end

        if status.available == false then
            DmRift.Text = "Rift: " .. tostring(status.reason)
            DmRift.TextWrapped = true
        elseif status.joinable then
            local left = math.max(0, (status.joinEndsAt or now) - now)
            DmRift.Text = string.format("Rift Portal: OPEN & JOINABLE (%s left) ⚡", SafeClock(left))
        elseif status.open then
            DmRift.Text = "Rift Portal: OPEN (In Progress) 🟢"
        else
            local nextIn = math.max(0, (status.nextOpenAt or (now + 600)) - now)
            DmRift.Text = string.format("Rift Portal: Closed (Next open in %s) ⏳", SafeClock(nextIn))
        end

        local cloverCd = SafeStoryCooldown(s.StoryCooldowns, "clover_capital_secret", now)
        local toyCd = SafeStoryCooldown(s.StoryCooldowns, "toy_kingdom_secret", now)
        local cloverTxt = cloverCd <= 0 and "READY 🟢" or ("CD " .. SafeClock(cloverCd))
        local toyTxt = toyCd <= 0 and "READY 🟢" or ("CD " .. SafeClock(toyCd))
        DmSecret.Text = string.format("Throne Room (Time Sage): %s\nToy Secret (Ruffy 4): %s", cloverTxt, toyTxt)
        DmSecret.TextColor3 = (cloverCd <= 0 or toyCd <= 0)
            and Color3.fromRGB(46, 204, 113)
            or Color3.fromRGB(255, 150, 180)
    end)
    if not ok then
        DmSlot1.Text = 'Monitor error: ' .. tostring(err)
        DmSlot1.TextWrapped = true
        DmSlot2.Text = 'Status update failed; see error above'
        DmSlot3.Text = 'Slot 3: status unavailable'
        DmRift.Text = 'Rift Portal: status unavailable'
        DmSecret.Text = 'Secret Bosses: status unavailable'
    end
end

-- Run immediately upon UI creation so labels NEVER stay on 'Checking...'
-- A single update loop avoids overlapping calls when a module yields.

task.spawn(function()
    while true do
        task.wait(1)
        UpdateDualMonitorUI()
    end
end)
